# 01.01.08 Pty

`fpl pty --session NAME` attaches to the kernel of the session NAME, starting it if there is none ([ADR-0006](../../02-docs/02.05-project/adr/ADR-0006-one-fpl-executable.md), [the design note](../../02-docs/02.05-project/notes/02.05.14-session-kernel.md)). The kernel is the seed's `fpl.kernel`, a child that outlives this process; the session's log lies beside its socket in `$XDG_STATE_HOME/fpl/sessions/` (default `~/.local/state/fpl/sessions/`), and the socket in `$XDG_RUNTIME_DIR/fpl/` when that is set, else beside the log. A name is `[A-Za-z0-9][A-Za-z0-9._-]*`.

With stdin a pipe, the whole of it is one `feed`, and the transcript of its inputs is printed: each input as it was entered, its results as `|=` lines, the notes as `;;` comments. `echo '1 2 +' | fpl pty --session x` is a complete call for an agent. The exit status is the worst of the turns: 0 ok, 1 an input failed, 2 refused. At a terminal the log is replayed first, then blocks of lines ended by a blank line are fed. `--replay` replays when piped too, `--watch` prints what other clients feed, `--stop` ends the kernel, `--fuel N` is the step budget of the call.

`Conn` and `attach` are also what `fpl tui` talks through.

```python {.marimo name="pty"}
"""The client of a session kernel, and the `fpl pty` command's work."""

import json
import os
import re
import socket
import subprocess
import time
from collections.abc import Iterator, Mapping
from pathlib import Path
from typing import Any, TextIO

from fpl.seed import command

NAME = re.compile(r"[A-Za-z0-9][A-Za-z0-9._-]*")
WAIT = 10.0
"""
Seconds a started kernel has to accept connections.
"""


def paths(name: str, env: Mapping[str, str]) -> tuple[Path, Path]:
    """
    The log and the socket of session `name`. Refused (ValueError): a
    name outside NAME.
    """
    if not NAME.fullmatch(name):
        raise ValueError(f"{name!r} is not a session name")
    state = env.get("XDG_STATE_HOME") or str(Path.home() / ".local/state")
    log = Path(state) / "fpl" / "sessions" / f"{name}.log"
    run = env.get("XDG_RUNTIME_DIR")
    folder = Path(run) / "fpl" if run else log.parent
    return log, folder / f"{name}.sock"


class Conn:
    """
    One connection to a kernel: requests out, responses in, as JSON
    lines. `ask` returns the response to a request; `events` the
    messages pushed after a `watch`.
    """

    def __init__(self, path: Path) -> None:
        self.sock = socket.socket(socket.AF_UNIX)
        self.sock.connect(str(path))
        self.reader = self.sock.makefile("rb")
        self.count = 0

    def read(self) -> dict[str, Any] | None:
        """
        The next message, or None when the kernel is gone.
        """
        line = self.reader.readline()
        return json.loads(line) if line else None

    def ask(self, op: str, **fields: Any) -> dict[str, Any]:
        """
        The response to a request. Refused (OSError): the kernel went
        away, or answered with an error.
        """
        self.count += 1
        request = {"op": op, "id": self.count, **fields}
        self.sock.sendall(json.dumps(request).encode() + b"\n")
        while (message := self.read()) is not None:
            if message.get("id") == self.count:
                if message["status"] != "done":
                    raise OSError(message["error"])
                return message
        raise ConnectionError("the kernel went away")

    def close(self) -> None:
        """
        Close the connection.
        """
        self.reader.close()
        self.sock.close()


def connect(path: Path) -> Conn | None:
    """
    A connection to the kernel at path, or None where none answers.
    """
    try:
        return Conn(path)
    except OSError:
        return None


def spawn(name: str, log: Path, path: Path, fuel: int | None) -> None:
    """
    Start the kernel of `name` as a child that outlives this process:
    a session of its own, no terminal, its complaints in `NAME.err`
    beside the log.
    """
    argv, env = command("fpl.kernel")
    argv += ["--name", name, "--log", str(log), "--socket", str(path)]
    if fuel is not None:
        argv += ["--fuel", str(fuel)]
    log.parent.mkdir(mode=0o700, parents=True, exist_ok=True)
    with log.with_suffix(".err").open("ab") as err:
        subprocess.Popen(  # noqa: S603 -- our own interpreter, argv a list
            argv,
            env=env,
            stdin=subprocess.DEVNULL,
            stdout=subprocess.DEVNULL,
            stderr=err,
            start_new_session=True,
        )


def attach(
    name: str, env: Mapping[str, str], fuel: int | None = None
) -> Conn:
    """
    A connection to the kernel of `name`, which is started if none
    answers. Refused (ValueError, LookupError, TimeoutError): a bad
    name, no seed, a kernel that does not come up within WAIT.
    """
    log, path = paths(name, env)
    found = connect(path)
    if found is not None:
        return found
    spawn(name, log, path, fuel)
    deadline = time.monotonic() + WAIT
    while time.monotonic() < deadline:
        found = connect(path)
        if found is not None:
            return found
        time.sleep(0.02)
    raise TimeoutError(f"no kernel for {name} at {path} (see {log.with_suffix('.err')})")


def who(env: Mapping[str, str]) -> dict[str, str]:
    """
    The model and session id this process runs under, as a feed names
    its author.
    """
    return {
        key: env[var]
        for key, var in (("model", "FPL_MODEL"), ("session", "FPL_SESSION_ID"))
        if var in env
    }


def fed(conn: Conn, text: str, env: Mapping[str, str], fuel: int | None) -> tuple[str, int]:
    """
    The transcript of `text` fed to the kernel, and the worst status of
    its turns.
    """
    extra = {} if fuel is None else {"fuel": fuel}
    got = conn.ask("feed", text=text, who=who(env), **extra)
    return got["transcript"], max((t["code"] for t in got["turns"]), default=0)


def blocks(stdin: TextIO) -> Iterator[str]:
    """
    The input of a terminal as blocks of lines, each ended by a blank
    line or the end.
    """
    block: list[str] = []
    for line in stdin:
        if line.strip():
            block.append(line)
        elif block:
            yield "".join(block)
            block = []
    if block:
        yield "".join(block)


def watching(conn: Conn, out: TextIO) -> int:
    """
    Print what other clients feed until the kernel goes or the user
    interrupts.
    """
    conn.ask("watch")
    try:
        while (message := conn.read()) is not None:
            out.write(message.get("transcript", ""))
            out.flush()
    except KeyboardInterrupt:
        pass
    return 0


def run(  # noqa: PLR0913 -- the command's options, one each
    name: str,
    stdin: TextIO,
    out: TextIO,
    err: TextIO,
    env: Mapping[str, str] = os.environ,
    *,
    replay: bool = False,
    watch: bool = False,
    stop: bool = False,
    fuel: int | None = None,
) -> int:
    """
    The `fpl pty` command: attach, then stop the kernel, watch, or feed
    the input. Its exit status; a refusal is one line on `err`, status 2.
    """
    try:
        if stop:
            found = connect(paths(name, env)[1])
            if found is not None:
                found.ask("stop")
            return 0
        conn = attach(name, env, fuel)
        if watch:
            return watching(conn, out)
        if replay or stdin.isatty():
            out.write(conn.ask("replay")["transcript"])
        worst = 0
        for text in blocks(stdin) if stdin.isatty() else [stdin.read()]:
            shown, code = fed(conn, text, env, fuel)
            out.write(shown)
            out.flush()
            worst = max(worst, code)
        return worst
    except (OSError, ValueError, LookupError, TimeoutError) as error:
        err.write(f"fpl pty: {error}\n")
        return 2
```
