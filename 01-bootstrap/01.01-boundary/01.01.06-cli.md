# 01.01.06 CLI

`python -m fpl` and the installed `fpl` script (`entry`, declared in `pyproject.toml`) land here ([ADR-0006](../../02-docs/02.05-project/adr/ADR-0006-one-fpl-executable.md)). The commands are read off the registries, so a new pass or composition needs no line here.

`fpl repl` opens the interactive interface, and `fpl` alone is short for it. For now that is the seed walker's REPL, `python -m fpl.repl` run from [the seed](../fpl/repl.py) in a child process ([01.01.07-seed](01.01.07-seed.md)): the tangled tree has no REPL of its own yet, and the seed cannot be imported beside it, since both packages are named `fpl`. The wheel carries the seed as `fpl/_seed/fpl` (`pyproject.toml`), so an installed `fpl` finds it there; a checkout finds it in `01-bootstrap/` (TD-0014).

`fpl pty --session NAME` joins the kernel of a named session, one process holding its log that any number of frontends attach to ([01.01.08-pty](01.01.08-pty.md), [the design note](../../02-docs/02.05-project/notes/02.05.14-session-kernel.md)). `fpl tui` is the full-screen interface on such a session ([01.01.09-tui](01.01.09-tui.md)); it needs the `tui` extra.

`fpl <stage>` is one filter across the pipe of [01.01.03-pipe](01.01.03-pipe.md): a pass of `PASSES`, or one of the two `ENDS`, `source` and `render`. `fpl run <file>` runs the whole chain in-process; `fpl <file>` is short for it, unless the file is named like a command.

`fpl graph <composition>` prints a composition as data ([01.04.05-graph](../01.04-compose/01.04.05-graph.md)): as text, Graphviz or Mermaid. `fpl shell <composition>` prints the shell pipe that runs it. The compositions are `COMPOSITIONS` of [01.04.02-run](../01.04-compose/01.04.02-run.md).

Click stays in this module: `import fpl` does not reach it, so the website, which has no Click, still imports the library.

```python {.marimo name="cli"}
"""`fpl repl`, or `fpl`, opens the REPL. `fpl <stage>` is one filter."""

import os
import subprocess
import sys
from pathlib import Path

import click

from fpl import pty, seed
from fpl.graph import dot, explain, mermaid
from fpl.pipe import ENDS, PASSES, cli, shell_script
from fpl.run import COMPOSITIONS, run_file

FORMATS = {"text": explain, "dot": dot, "mermaid": mermaid}
"""
The ways `fpl graph` writes a composition down.
"""


def repl() -> int:
    """
    Run the seed REPL on this terminal; its exit status.
    """
    try:
        argv, env = seed.command("fpl.repl")
    except LookupError as missing:
        print(f"fpl: no REPL: no seed in {missing}", file=sys.stderr)
        return 2
    return subprocess.run(argv, env=env, check=False).returncode


@click.group(invoke_without_command=True)
@click.pass_context
def main(ctx: click.Context) -> None:
    """Open the REPL, run a file, or run one pass across a pipe."""
    if ctx.invoked_subcommand is None:
        ctx.invoke(repl_command)


@main.command("repl")
def repl_command() -> None:
    """Open the REPL. Bare `fpl` does the same."""
    sys.exit(repl())


@main.command("pty")
@click.option("--session", "name", default="default", show_default=True)
@click.option("--replay", is_flag=True, help="Print the log first.")
@click.option("--watch", is_flag=True, help="Print what other clients feed.")
@click.option("--stop", is_flag=True, help="End the session's kernel.")
@click.option("--fuel", type=click.IntRange(min=0), help="Step budget.")
@click.option("--at", help="Read the program as of this event; append nothing.")
@click.option("--head", help="Append only if this event is still the head.")
def pty_command(  # noqa: PLR0913 -- one parameter per option
    name: str,
    replay: bool,
    watch: bool,
    stop: bool,
    fuel: int | None,
    at: str | None,
    head: str | None,
) -> None:
    """Join the named session's kernel: stdin in, the transcript out."""
    sys.exit(
        pty.run(
            name,
            sys.stdin,
            sys.stdout,
            sys.stderr,
            replay=replay,
            watch=watch,
            stop=stop,
            fuel=fuel,
            at=at,
            head=head,
        )
    )


def tui_main(name: str) -> int:
    """
    Open the full-screen interface on session `name`: refused, with one
    line and status 2, where there is no terminal to draw on or Textual
    is not installed.
    """
    if not (sys.stdin.isatty() and sys.stdout.isatty()) or os.environ.get(
        "TERM", "dumb"
    ) in ("dumb", ""):
        print(
            "fpl tui: needs a terminal that can draw; use `fpl repl`, "
            "or `fpl pty` from a script",
            file=sys.stderr,
        )
        return 2
    try:
        from fpl import tui  # noqa: PLC0415 -- Textual loads for this command only
    except ImportError:
        print(
            "fpl tui: Textual is not installed: pip install 'fpl[tui]'",
            file=sys.stderr,
        )
        return 2
    return tui.run(name, os.environ)


@main.command("tui")
@click.option("--session", "name", default="tui", show_default=True)
def tui_command(name: str) -> None:
    """Open the full-screen interface on the named session."""
    sys.exit(tui_main(name))


@main.command("run")
@click.argument("file", type=click.Path(exists=True, dir_okay=False))
def run_command(file: str) -> None:
    """Run a file in-process. ABTs stay in this process."""
    sys.exit(run_file(file))


COMPOSITION = click.argument("composition", type=click.Choice(COMPOSITIONS))


@main.command("graph")
@COMPOSITION
@click.option(
    "--format", "form", type=click.Choice(FORMATS), default="text"
)
def graph_command(composition: str, form: str) -> None:
    """Print a composition as a graph. Nothing runs."""
    click.echo(FORMATS[form](COMPOSITIONS[composition]).rstrip("\n"))


@main.command("shell")
@COMPOSITION
def shell_command(composition: str) -> None:
    """Print the shell pipe that runs a composition."""
    click.echo(shell_script(COMPOSITIONS[composition]))


def _filter(name: str, summary: str) -> click.Command:
    """
    The filter command `name`: one stage from stdin to stdout.
    """

    def command() -> None:
        sys.exit(cli(name))

    return click.command(name, help=summary)(command)


for _name, _end in ENDS.items():
    main.add_command(_filter(_name, (_end.__doc__ or "").strip()))
for _name, _pass in PASSES.items():
    main.add_command(
        _filter(_name, f"FON<{_pass.source}> in, FON<{_pass.target}> out.")
    )


def dispatch(argv: list[str]) -> int:
    """
    `fpl <file>` is `fpl run <file>`; anything else is Click.

    A command name wins over a file of the same name.
    """
    if len(argv) == 1 and argv[0] not in main.commands:
        if Path(argv[0]).is_file():
            argv = ["run", *argv]
    try:
        main(args=argv, standalone_mode=False)
    except SystemExit as stop:
        return int(stop.code or 0)
    except click.ClickException as refused:
        refused.show()
        return refused.exit_code
    return 0


def entry() -> None:
    """The `fpl` console script: exit with the dispatch status."""
    sys.exit(dispatch(sys.argv[1:]))


if __name__ == "__main__":
    entry()
```
